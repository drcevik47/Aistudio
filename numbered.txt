   171	    private suspend fun silentRefresh(cycleCount: Int = 0) {
   172	        try {
   173	            // Periodically check and sync any unfilled orders in DB with exchange
   174	            if (cycleCount % 4 == 0) {
   175	                repository.syncUnfilledOrdersWithExchange()
   176	            }
   177	    
   178	            // Periodically prune logs older than 24 hours or exceeding 1000 items
   179	            if (cycleCount % 30 == 0) {
   180	                repository.pruneLogs()
   181	            }
   182	    
   183	            var currentPrice = _uiState.value.currentPrice
   184	            var priceChange = _uiState.value.price24hChange
   185	            val tickerRes = repository.getTicker()
   186	            tickerRes.onSuccess { ticker ->
   187	                currentPrice = ticker.currentPrice
   188	                priceChange = ticker.changePercent24h
   189	            }
   190	    
   191	
   192	            var usdt = _uiState.value.usdtBalance
   193	            var baseQty = _uiState.value.baseCoinBalance
   194	            // Stagger balance fetch every ~8 seconds (every 2 cycles) or on initial load
   195	            if (cycleCount % 2 == 0 || usdt <= 0.0) {
   196	                val balanceRes = repository.getWalletBalance()
   197	                balanceRes.onSuccess { map ->
   198	                    usdt = map["USDT"] ?: 0.0
   199	                    baseQty = map[_uiState.value.activeBaseCoin] ?: 0.0
   200	                }
   201	            }
   202	        
   203	    
   204	
   205	            var openOrders = _uiState.value.activeOrders
   206	            // Stagger open orders fetch every ~8 seconds (alternating cycle)
   207	            if (cycleCount % 2 == 1 || openOrders.isEmpty()) {
   208	                val openOrdersRes = repository.getOpenOrders()
   209	                openOrdersRes.onSuccess { list ->
   210	                    openOrders = list
   211	                }
   212	            }
   213	        
   214	    
   215	
   216	            val analysis = RebalanceEngine.analyzePortfolio(
   217	                usdtBalance = usdt,
   218	                baseCoinBalance = baseQty,
   219	                currentPrice = currentPrice
   220	            )
   221	
   222	            val anchorBasePrice = if (preferences.isBotActive && preferences.lastRebalancePrice > 0.0) {
   223	                preferences.lastRebalancePrice
   224	     else {
   225	                currentPrice
   226	    
   227	
   228	            val gridPlan = RebalanceEngine.calculateGridOrders(
   229	                usdtBalance = usdt,
   230	                baseCoinBalance = baseQty,
   231	                basePrice = anchorBasePrice,
   232	                stepPercent = _uiState.value.stepPercent
   233	            )
   234	
   235	            _uiState.update {
   236	                it.copy(
   237	                    currentPrice = currentPrice,
   238	                    price24hChange = priceChange,
   239	                    usdtBalance = usdt,
   240	                    baseCoinBalance = baseQty,
   241	                    activeOrders = openOrders,
   242	                    portfolioAnalysis = analysis,
   243	                    gridPlan = gridPlan,
   244	                    isBotActive = preferences.isBotActive,
   245	                    lastRebalancePrice = preferences.lastRebalancePrice
   246	                )
   247	            }
   248	                )
   249	    
   250	 catch (e: Exception) {
   251	            // Silently ignore background polling exceptions
   252	
   253	    }

   444	     * Executes the initial 50/50 balance trade upon user confirmation.
   445	     */
   446	    fun executeInitialRebalance() {
   447	        val analysis = _uiState.value.portfolioAnalysis ?: return
   448	        val currentPrice = _uiState.value.currentPrice
   449	        if (currentPrice <= 0.0) return
   450	
   451	        viewModelScope.launch(Dispatchers.IO) {
   452	            _uiState.update { it.copy(isExecutingRebalance = true, errorMessage = null) }
   453	
   454	            val side = if (analysis.requiredAction == RebalanceAction.BUY_BASE) "Buy" else "Sell"
   455	            val qty = analysis.deltaBase
   456	
   457	            repository.log(
   458	                LogLevel.INFO,
   459	                "InitialRebalance",
   460	                "Kullanıcı onayı ile başlangıç eşitlemesi yapılıyor: $side ${RebalanceEngine.format4(qty)} ${_uiState.value.activeBaseCoin} (~${RebalanceEngine.format2(analysis.deltaUsdt)} USDT)"
   461	            )
   462	
   463	            // Execute Market order for instant 50/50 equalization
   464	            val res = repository.createOrder(
   465	                side = side,
   466	                orderType = "Market",
   467	                qty = qty,
   468	                price = null,
   469	                triggerReason = "InitialRebalance"
   470	            )
   471	
   472	            res.onSuccess { orderId ->
   473	                repository.log(LogLevel.SUCCESS, "InitialRebalance", "Başlangıç %50-%50 eşitleme emri gerçekleşti: $orderId")
   474	                preferences.lastRebalancePrice = currentPrice
   475	                repository.recordOrderFilled(
   476	                    orderId = orderId,
   477	                    side = side,
   478	                    price = currentPrice,
   479	                    qty = qty,
   480	                    triggerReason = "InitialRebalance"
   481	                )
   482	                _uiState.update {
   483	                    it.copy(
   484	                        isExecutingRebalance = false,
   485	                        showInitialRebalanceDialog = false,
   486	                        lastRebalancePrice = currentPrice,
   487	                        statusMessage = "Portföy %50-%50 olarak başarıyla dengelendi!"
   488	                    )
   489	        
   490	                kotlinx.coroutines.delay(1000)
   491	                refreshData()
   492	    .onFailure { err ->
   493	                repository.log(LogLevel.ERROR, "InitialRebalance", "Eşitleme emri verilemedi: ${err.message}")
   494	                _uiState.update {
   495	                    it.copy(
   496	                        isExecutingRebalance = false,
   497	                        errorMessage = "Eşitleme emri hatası: ${err.message}"
   498	                    )
   499	        
   500	    
   501	
   502	    }

   504	    /**
   505	     * Starts 24/7 background Bot Service and places grid orders (+2% sell & -2% buy).
   506	     */
   507	    fun startBot() {
   508	        viewModelScope.launch(Dispatchers.IO) {
   509	            _uiState.update { it.copy(isLoading = true, errorMessage = null) }
   510	
   511	            val state = _uiState.value
   512	            val price = state.currentPrice
   513	            val usdt = state.usdtBalance
   514	            val baseQty = state.baseCoinBalance
   515	
   516	            if (price <= 0.0 || usdt <= 0.0 || baseQty <= 0.0) {
   517	                _uiState.update {
   518	                    it.copy(
   519	                        isLoading = false,
   520	                        errorMessage = "Botu başlatmak için ${_uiState.value.activeBaseCoin} ve USDT bakiyesi gereklidir."
   521	                    )
   522	        
   523	                return@launch
   524	    
   525	
   526	            val plan = RebalanceEngine.calculateGridOrders(
   527	                usdtBalance = usdt,
   528	                baseCoinBalance = baseQty,
   529	                basePrice = price,
   530	                stepPercent = state.stepPercent
   531	            )
   532	
   533	            if (!plan.isValid) {
   534	                _uiState.update {
   535	                    it.copy(
   536	                        isLoading = false,
   537	                        errorMessage = plan.validationMessage
   538	                    )
   539	        
   540	                return@launch
   541	    
   542	
   543	            // Cancel any old orders first
   544	            repository.cancelAllBaseOrders()
   545	
   546	            // Place Limit Sell (+2%)
   547	            val sellRes = repository.createOrder(
   548	                side = "Sell",
   549	                orderType = "Limit",
   550	                qty = plan.sellBaseQty,
   551	                price = plan.sellLimitPrice,
   552	                triggerReason = "GridStepUpSell"
   553	            )
   554	
   555	            // Place Limit Buy (-2%)
   556	            val buyRes = repository.createOrder(
   557	                side = "Buy",
   558	                orderType = "Limit",
   559	                qty = plan.buyBaseQty,
   560	                price = plan.buyLimitPrice,
   561	                triggerReason = "GridStepDownBuy"
   562	            )
   563	
   564	            val orderError = sellRes.exceptionOrNull()?.message ?: buyRes.exceptionOrNull()?.message
   565	            if (orderError != null) {
   566	                _uiState.update {
   567	                    it.copy(
   568	                        isLoading = false,
   569	                        errorMessage = "Emir iletilemedi: $orderError"
   570	                    )
   571	        
   572	                return@launch
   573	    
   574	
   575	            sellRes.onSuccess { sellId ->
   576	                preferences.activeSellOrderId = sellId
   577	    
   578	            buyRes.onSuccess { buyId ->
   579	                preferences.activeBuyOrderId = buyId
   580	    
   581	
   582	            preferences.isBotActive = true
   583	            preferences.lastRebalancePrice = price
   584	            repository.lastGridOrderPlacedTimeMs = System.currentTimeMillis()
   585	
   586	            // Start Foreground 24/7 Service
   587	            TradingBotService.start(getApplication())
   588	
   589	            _uiState.update {
   590	                it.copy(
   591	                    isLoading = false,
   592	                    isBotActive = true,
   593	                    statusMessage = "7/24 Dengeleme Botu Başlatıldı! %${state.stepPercent} Limit emirler aktifleştirildi."
   594	                )
   595	    
   596	            refreshData()
   597	
   598	    }

   739	    fun exportKlines(context: android.content.Context, symbol: String, interval: String, format: String, startTime: Long, endTime: Long) {
   740	        viewModelScope.launch {
   741	            try {
   742	                val effectiveSymbol = if (symbol == "ALL" || symbol.isBlank()) "${_uiState.value.activeSymbol}" else symbol
   743	                android.widget.Toast.makeText(context, "Veriler indiriliyor... Lütfen bekleyin.", android.widget.Toast.LENGTH_SHORT).show()
   744	                
   745	                var currentEnd: Long = endTime
   746	                val allKlinesList = mutableListOf<List<String>>()
   747	                var apiError: String? = null
   748	                val limitToFetch = 1000
   749	                
   750	                while (true) {
   751	                    val result = repository.getKlines(symbol = effectiveSymbol, interval = interval, start = startTime, end = currentEnd, limit = limitToFetch)
   752	                    
   753	                    if (result.isSuccess) {
   754	                        val klineResult = result.getOrNull()
   755	                        val data = klineResult?.list ?: emptyList()
   756	                        
   757	                        if (data.isEmpty()) break
   758	                        
   759	                        // Bybit returns newest first. Let's filter just in case they return out of range.
   760	                        val validData = data.filter { k -> 
   761	                            val ts = k[0].toLongOrNull() ?: 0L
   762	                            ts >= startTime && ts <= currentEnd
   763	                
   764	                        
   765	                        if (validData.isEmpty()) break
   766	                        
   767	                        allKlinesList.addAll(validData)
   768	                        
   769	                        if (data.size < limitToFetch) {
   770	                            // Reached the oldest available data
   771	                            break
   772	                
   773	                        
   774	                        // The oldest candle is the last one in the list
   775	                        val oldestTimestamp = data.last()[0].toLongOrNull()
   776	                        if (oldestTimestamp != null) {
   777	                            if (oldestTimestamp <= startTime) break
   778	                            currentEnd = oldestTimestamp - 1
   779	                 else {
   780	                            break
   781	                
   782	             else {
   783	                        apiError = result.exceptionOrNull()?.message ?: "Bilinmeyen hata"
   784	                        break
   785	            
   786	                    
   787	                    // Small delay to prevent rate limit
   788	                    kotlinx.coroutines.delay(250)
   789	        
   790	                
   791	                if (allKlinesList.isEmpty()) {
   792	                    if (apiError != null) {
   793	                        android.widget.Toast.makeText(context, "Veri alınamadı: $apiError", android.widget.Toast.LENGTH_LONG).show()
   794	             else {
   795	                        android.widget.Toast.makeText(context, "Seçilen tarih aralığında veri bulunamadı", android.widget.Toast.LENGTH_SHORT).show()
   796	            
   797	         else {
   798	                    val combinedResult = com.example.data.remote.model.KlineResult(
   799	                        symbol = effectiveSymbol,
   800	                        category = "spot",
   801	                        list = allKlinesList
   802	                    )
   803	                    com.example.ui.components.KlineExporter.shareKlinesAsFile(context, combinedResult, effectiveSymbol, interval, format)
   804	        
   805	     catch (e: Exception) {
   806	                android.util.Log.e("KlineExport", "Error exporting klines", e)
   807	                android.widget.Toast.makeText(context, "Veri alınırken hata oluştu: ${e.message}", android.widget.Toast.LENGTH_LONG).show()
   808	    
   809	
   810	    }
