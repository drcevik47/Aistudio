   173	    private suspend fun silentRefresh(cycleCount: Int = 0) {
   174	        try {
   175	            if (cycleCount % 4 == 0) {
   176	                repository.syncUnfilledOrdersWithExchange()
   177	            }
   178	            if (cycleCount % 30 == 0) {
   179	                repository.pruneLogs()
   180	            }
   181	
   182	            var currentPrice = _uiState.value.currentPrice
   183	            var priceChange = _uiState.value.price24hChange
   184	            
   185	            val tickerRes = repository.getTicker()
   186	            tickerRes.onSuccess { ticker ->
   187	                currentPrice = ticker.currentPrice
   188	                priceChange = ticker.changePercent24h
   189	            }
   190	
   191	            var usdt = _uiState.value.usdtBalance
   192	            var baseQty = _uiState.value.baseCoinBalance
   193	
   194	            if (cycleCount % 2 == 0 || usdt <= 0.0) {
   195	                val balanceRes = repository.getWalletBalance()
   196	                balanceRes.onSuccess { map ->
   197	                    usdt = map["USDT"] ?: 0.0
   198	                    baseQty = map[_uiState.value.activeBaseCoin] ?: 0.0
   199	                }
   200	            }
   201	
   202	            var openOrders = _uiState.value.activeOrders
   203	            if (cycleCount % 2 == 1 || openOrders.isEmpty()) {
   204	                val openOrdersRes = repository.getOpenOrders()
   205	                openOrdersRes.onSuccess { list ->
   206	                    openOrders = list
   207	                }
   208	            }
   209	
   210	            val analysis = RebalanceEngine.analyzePortfolio(
   211	                usdtBalance = usdt,
   212	                baseCoinBalance = baseQty,
   213	                currentPrice = currentPrice
   214	            )
   215	
   216	            val anchorBasePrice = if (preferences.isBotActive && preferences.lastRebalancePrice > 0.0) {
   217	                preferences.lastRebalancePrice
   218	            } else {
   219	                currentPrice
   220	            }
   221	
   222	            val gridPlan = RebalanceEngine.calculateGridOrders(
   223	                usdtBalance = usdt,
   224	                baseCoinBalance = baseQty,
   225	                basePrice = anchorBasePrice,
   226	                stepPercent = _uiState.value.stepPercent
   227	            )
   228	
   229	            _uiState.update {
   230	                it.copy(
   231	                    currentPrice = currentPrice,
   232	                    price24hChange = priceChange,
   233	                    usdtBalance = usdt,
   234	                    baseCoinBalance = baseQty,
   235	                    activeOrders = openOrders,
   236	                    portfolioAnalysis = analysis,
   237	                    gridPlan = gridPlan,
   238	                    isBotActive = preferences.isBotActive,
   239	                    lastRebalancePrice = preferences.lastRebalancePrice
   240	                )
   241	            }
   242	        } catch (e: Exception) {
   243	            // Silently ignore background polling exceptions
   244	        }
   245	    }

   389	        }

   392	    private fun recalculateGridPlan() {
   393	        val state = _uiState.value
   394	        val anchorBasePrice = if (preferences.isBotActive && preferences.lastRebalancePrice > 0.0) {
   395	            preferences.lastRebalancePrice
   396	        } else {
   397	            state.currentPrice
   398	        }
   399	        val plan = RebalanceEngine.calculateGridOrders(
   400	            usdtBalance = state.usdtBalance,
   401	            baseCoinBalance = state.baseCoinBalance,
   402	            basePrice = anchorBasePrice,
   403	            stepPercent = state.stepPercent
   404	        )
   405	        _uiState.update { it.copy(gridPlan = plan) }
   406	    }

   416	    fun executeInitialRebalance() {
   417	        val analysis = _uiState.value.portfolioAnalysis ?: return
   418	        val currentPrice = _uiState.value.currentPrice
   419	        if (currentPrice <= 0.0) return
   420	
   421	        viewModelScope.launch(Dispatchers.IO) {
   422	            _uiState.update { it.copy(isLoading = true, isExecutingRebalance = true, showInitialRebalanceDialog = false) }
   423	
   424	            val side = if (analysis.requiredAction == RebalanceAction.BUY_BASE) "Buy" else "Sell"
   425	            val qty = analysis.deltaBase
   426	
   427	            val res = repository.createOrder(
   428	                side = side,
   429	                orderType = "Market",
   430	                qty = qty,
   431	                price = null
   432	            )
   433	
   434	            res.fold(
   435	                onSuccess = { orderId ->
   436	                    repository.log(LogLevel.INFO, "Rebalance", "İlk dengeleme Market emri (${side} ${qty}) başarıyla iletildi.")
   437	                    _uiState.update {
   438	                        it.copy(
   439	                            isLoading = false,
   440	                            isExecutingRebalance = false,
   441	                            statusMessage = "Dengeleme işlemi ($side ${RebalanceEngine.format4(qty)}) başarıyla gerçekleşti."
   442	                        )
   443	                    }
   444	                    startBot()
   445	                },
   446	                onFailure = { err ->
   447	                    repository.log(LogLevel.ERROR, "Rebalance", "İlk dengeleme başarısız: ${err.message}")
   448	                    _uiState.update {
   449	                        it.copy(
   450	                            isLoading = false,
   451	                            isExecutingRebalance = false,
   452	                            errorMessage = "Dengeleme başarısız: ${err.message}"
   453	                        )
   454	                    }
   455	                }
   456	            )
   457	            refreshData()
   458	        }
   459	    }

   576	        viewModelScope.launch {
   577	            _tradeAnalysis.update {
   578	                it.copy(
   579	                    isLoading = true,
   580	                    errorMessage = null,
   581	                    syncNotice = null,
   582	                    progressText = "Borsadan işlemler çekiliyor ve veritabanı kontrol ediliyor..."
   583	                )
   584	            }
   585	            val result = repository.syncTradesFromExchange(
   586	                symbol = symbol,
   587	                daysBack = daysBack,
   588	                startTimestamp = startTimestamp,
   589	                onProgress = { currentWindow, totalWindows, fetchedCount ->
   590	                    _tradeAnalysis.update {
   591	                        it.copy(progressText = "$currentWindow / $totalWindows hafta tarandı ($fetchedCount işlem)")
   592	                    }
   593	                }
   594	            )
   595	            result.onSuccess { syncRes ->
   596	                val notice = if (syncRes.newlyAddedCount > 0) {
   597	                    "✓ ${syncRes.totalFetched} işlem tarandı. ${syncRes.newlyAddedCount} YENİ işlem veritabanına eklendi! (Toplam DB: ${syncRes.totalInDb})"
   598	                } else {
   599	                    "✓ ${syncRes.totalFetched} işlem tarandı. Tüm işlemler zaten veritabanında kayıtlı (Yeni işlem yok)."
   600	                }
   601	                _tradeAnalysis.update {
   602	                    it.copy(
   603	                        isLoading = false,
   604	                        analysis = syncRes.analysis,
   605	                        syncResult = syncRes,
   606	                        syncNotice = notice,
   607	                        errorMessage = null,
   608	                        lastFetchedAt = System.currentTimeMillis(),
   609	                        progressText = null
   610	                    )
   611	                }
   612	            }.onFailure { err ->
   613	                _tradeAnalysis.update {
   614	                    it.copy(
   615	                        isLoading = false,
   616	                        errorMessage = err.message ?: "İşlem geçmişi çekilemedi",
   617	                        progressText = null
   618	                    )
   619	                }
   620	            }
   621	        }
   622	    }

   624	    fun exportKlines(context: android.content.Context, symbol: String, interval: String, format: String, startTime: Long, endTime: Long) {
   625	        viewModelScope.launch {
   626	            try {
   627	                val effectiveSymbol = if (symbol == "ALL" || symbol.isBlank()) _uiState.value.activeSymbol else symbol
   628	                android.widget.Toast.makeText(context, "Veriler indiriliyor... Lütfen bekleyin.", android.widget.Toast.LENGTH_SHORT).show()
   629	                
   630	                var currentEnd: Long = endTime
   631	                val allKlinesList = mutableListOf<List<String>>()
   632	                var apiError: String? = null
   633	                val limitToFetch = 1000
   634	                
   635	                while (true) {
   636	                    val result = repository.getKlines(symbol = effectiveSymbol, interval = interval, start = startTime, end = currentEnd, limit = limitToFetch)
   637	                    
   638	                    if (result.isSuccess) {
   639	                        val klineResult = result.getOrNull()
   640	                        val data = klineResult?.list ?: emptyList()
   641	                        
   642	                        if (data.isEmpty()) break
   643	                        
   644	                        val validData = data.filter { k -> 
   645	                            val ts = k[0].toLongOrNull() ?: 0L
   646	                            ts >= startTime && ts <= currentEnd
   647	                        }
   648	                        
   649	                        if (validData.isEmpty()) break
   650	                        
   651	                        allKlinesList.addAll(validData)
   652	                        
   653	                        if (data.size < limitToFetch) break
   654	                        
   655	                        val oldestTimestamp = data.last()[0].toLongOrNull()
   656	                        if (oldestTimestamp != null) {
   657	                            if (oldestTimestamp <= startTime) break
   658	                            currentEnd = oldestTimestamp - 1
   659	                        } else {
   660	                            break
   661	                        }
   662	                    } else {
   663	                        apiError = result.exceptionOrNull()?.message ?: "Bilinmeyen hata"
   664	                        break
   665	                    }
   666	                    
   667	                    kotlinx.coroutines.delay(250)
   668	                }
   669	                
   670	                if (allKlinesList.isEmpty()) {
   671	                    if (apiError != null) {
   672	                        android.widget.Toast.makeText(context, "Veri alınamadı: $apiError", android.widget.Toast.LENGTH_LONG).show()
   673	                    } else {
   674	                        android.widget.Toast.makeText(context, "Seçilen tarih aralığında veri bulunamadı", android.widget.Toast.LENGTH_SHORT).show()
   675	                    }
   676	                } else {
   677	                    val combinedResult = com.example.data.remote.model.KlineResult(
   678	                        symbol = effectiveSymbol,
   679	                        category = "spot",
   680	                        list = allKlinesList
   681	                    )
   682	                    com.example.ui.components.KlineExporter.shareKlinesAsFile(context, combinedResult, effectiveSymbol, interval, format)
   683	                }
   684	            } catch (e: Exception) {
   685	                android.util.Log.e("KlineExport", "Error exporting klines", e)
   686	                android.widget.Toast.makeText(context, "Veri alınırken hata oluştu: ${e.message}", android.widget.Toast.LENGTH_LONG).show()
   687	            }
   688	        }
   689	    }
